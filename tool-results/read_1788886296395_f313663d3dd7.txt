   331→function BlogForm({ id, onBack }: { id: string | null; onBack: () => void }) {
   332→  const [currentId, setCurrentId] = useState<string | null>(id);
   333→  const isNew = !currentId;
   334→  const { data: post, isLoading } = useAdminBlog(currentId);
   335→  const { data: blogCategories } = useBlogCategoriesList();
   336→  const { data: products } = useAdminProducts({ page: 1, pageSize: 200, q: "", category: "", status: "published" });
   337→  const { data: categories } = useAdminCategories();
   338→  const { data: allBlogs } = useAdminBlogs({ page: 1, pageSize: 200, q: "", status: "all" });
   339→  const { data: allFaqs } = useAdminFaqs("");
   340→
   341→  const [form, setForm] = useState<BlogFormState>(EMPTY_BLOG);
   342→  const [initialFaqIds, setInitialFaqIds] = useState<string[]>([]);
   343→  const [submitted, setSubmitted] = useState(false);
   344→  const [saving, setSaving] = useState(false);
   345→  const [tab, setTab] = useState("content");
   346→  const [relatedQ, setRelatedQ] = useState("");
   347→  const initRef = useRef<string | null>(null);
   348→  const { toast } = useToast();
   349→  const qc = useQueryClient();
   350→
   351→  const blogFaqs = useMemo(
   352→    () => (allFaqs ?? []).filter((f) => f.entityType === "BLOG" && f.entityId === currentId),
   353→    [allFaqs, currentId],
   354→  );
   355→
   356→  useEffect(() => {
   357→    if (!post || initRef.current === post.id) return;
   358→    initRef.current = post.id;
   359→    const faqs = (allFaqs ?? [])
   360→      .filter((f) => f.entityType === "BLOG" && f.entityId === post.id)
   361→      .map((f) => ({ id: f.id, question: f.question, answer: f.answer }));
   362→    setInitialFaqIds((allFaqs ?? []).filter((f) => f.entityType === "BLOG" && f.entityId === post.id).map((f) => f.id));
   363→    setForm({
   364→      title: post.title,
   365→      slug: post.slug,
   366→      excerpt: post.excerpt,
   367→      content: post.content,
   368→      coverImage: post.coverImage,
   369→      coverAlt: post.coverAlt,
   370→      author: post.author,
   371→      readingTime: post.readingTime,
   372→      tags: post.tags,
   373→      featured: post.featured,
   374→      status: post.status,
   375→      publishAt: toDatetimeLocal(post.publishedAt),
   376→      blogCategoryId: post.blogCategorySlug ? String(post.blogCategorySlug) : "",
   377→      relatedProductSlugs: safeParseSlugs(post.relatedProductSlugs),
   378→      faqs,
   379→      seoTitle: post.seoTitle,
   380→      metaDescription: post.metaDescription,
   381→      focusKeyword: post.focusKeyword,
   382→      secondaryKeywords: post.secondaryKeywords,
   383→      ogTitle: post.ogTitle,
   384→      ogDescription: post.ogDescription,
   385→      canonicalUrl: post.canonicalUrl,
   386→    });
   387→  }, [post, allFaqs]);
   388→
   389→  // blogCategoryId: we keep slug in state but need id at save; map slug→id
   390→  const blogCatIdFromSlug = useMemo(() => {
   391→    if (!form.blogCategoryId) return "";
   392→    const cat = (blogCategories ?? []).find((c) => c.slug === form.blogCategoryId);
   393→    return cat?.id ?? "";
   394→  }, [form.blogCategoryId, blogCategories]);

   396→  function set<K extends keyof BlogFormState>(key: K, value: BlogFormState[K]) {
   397→    setForm((f) => ({ ...f, [key]: value }));

   400→  function estimateReadingTime() {
   401→    const words = form.content
   402→      .replace(/<[^>]+>/g, " ")
   403→      .split(/\s+/)
   404→      .filter(Boolean).length;
   405→    const minutes = Math.max(1, Math.round(words / 200));
   406→    set("readingTime", minutes);
   407→    toast({ title: `Reading time set to ${minutes} min`, description: `${words} words at an average reading pace.` });
   408→  }

   410→  function generateSeoDefaults() {
   411→    setForm((f) => ({
   412→      ...f,
   413→      seoTitle: f.seoTitle || `${f.title || "Blog post"} | Artistic by Khushi`,
   414→      metaDescription: f.metaDescription || f.excerpt.slice(0, 155),
   415→      ogTitle: f.ogTitle || `${f.title} | Artistic by Khushi`,
   416→      ogDescription: f.ogDescription || f.excerpt.slice(0, 155),
   417→    }));
   418→    toast({ title: "SEO defaults filled" });
   419→  }

   421→  // internal link options for the editor

   423→    const links: InternalLinkOption[] = [];
   424→    for (const p of products?.items ?? []) links.push({ label: p.name, href: `#/product/${p.slug}`, group: "Products" });
   425→    for (const c of categories ?? []) links.push({ label: c.name, href: `#/category/${c.slug}`, group: "Categories" });
   426→    for (const b of allBlogs?.items ?? []) if (b.id !== currentId) links.push({ label: b.title, href: `#/blog/${b.slug}`, group: "Blog posts" });
   427→    return links;
   428→  }, [products, categories, allBlogs, currentId]);
   429→
   430→  const relatedProductOptions = useMemo(() => {
   431→    const list = products?.items ?? [];
   432→    const needle = relatedQ.trim().toLowerCase();
   433→    return needle ? list.filter((p) => p.name.toLowerCase().includes(needle)) : list;
   434→  }, [products, relatedQ]);

   436→  async function save() {
   437→    setSubmitted(true);
   438→    if (!form.title.trim()) {
   439→      setTab("content");
   440→      toast({ title: "Blog title is required", variant: "destructive" });
   441→      return;
   442→    }
   443→    if (form.status === "SCHEDULED" && !form.publishAt) {
   444→      setTab("publish");
   445→      toast({ title: "Pick a schedule date", description: "Scheduled posts need a publish date & time.", variant: "destructive" });
   446→      return;
   447→    }
   448→
   449→    setSaving(true);
   450→    try {
   451→      const body = {
   452→        title: form.title.trim(),
   453→        slug: form.slug,
   454→        excerpt: form.excerpt,
   455→        content: form.content,
   456→        coverImage: form.coverImage,
   457→        coverAlt: form.coverAlt,
   458→        author: form.author,
   459→        readingTime: form.readingTime,
   460→        tags: form.tags,
   461→        featured: form.featured,
   462→        status: form.status,
   463→        publishAt: form.status === "SCHEDULED" && form.publishAt ? new Date(form.publishAt).toISOString() : "",
   464→        blogCategoryId: blogCatIdFromSlug || null,
   465→        relatedProductSlugs: JSON.stringify(form.relatedProductSlugs),
   466→        seoTitle: form.seoTitle,
   467→        metaDescription: form.metaDescription,
   468→        focusKeyword: form.focusKeyword,
   469→        secondaryKeywords: form.secondaryKeywords,
   470→        ogTitle: form.ogTitle,
   471→        ogDescription: form.ogDescription,
   472→        canonicalUrl: form.canonicalUrl,
   473→      };
   474→      let blogId: string;
   475→      if (isNew) {
   476→        const res = await api.post<{ blog: PublicBlogPost }>("/api/admin/blogs", body);
   477→        blogId = res.blog.id;
   478→      } else {
   479→        const res = await api.put<{ blog: PublicBlogPost }>(`/api/admin/blogs/${currentId}`, body);
   480→        blogId = res.blog.id;
   481→      }
   482→
   483→      // FAQ sync
   484→      const keptIds = new Set(form.faqs.filter((f) => f.id).map((f) => f.id as string));
   485→      for (const faqId of initialFaqIds) {
   486→        if (!keptIds.has(faqId)) await api.delete(`/api/admin/faqs/${faqId}`);
   487→      }
   488→      for (const faq of form.faqs) {
   489→        if (!faq.question.trim() || !faq.answer.trim()) continue;
   490→        if (faq.id) {
   491→          await api.put(`/api/admin/faqs/${faq.id}`, { question: faq.question, answer: faq.answer });
   492→        } else {
   493→          await api.post("/api/admin/faqs", {
   494→            question: faq.question,
   495→            answer: faq.answer,
   496→            entityType: "BLOG",
   497→            entityId: blogId,
   498→            displayOrder: 0,
   499→            published: true,
   500→          });
   501→        }
   502→      }
   503→
   504→      toast({ title: "Blog post saved", description: form.title });
   505→      for (const key of PUBLIC_CACHE_KEYS) void qc.invalidateQueries({ queryKey: key });
   506→      void qc.invalidateQueries({ queryKey: ["admin", "blogs"] });
   507→      void qc.invalidateQueries({ queryKey: ["admin", "dashboard"] });
   508→      void qc.invalidateQueries({ queryKey: ["admin", "faqs"] });
   509→      setInitialFaqIds(form.faqs.filter((f) => f.id).map((f) => f.id as string));
   510→
   511→      if (isNew) setCurrentId(blogId);
   512→    } catch (e) {
   513→      toast({ title: "Save failed", description: errMsg(e), variant: "destructive" });
   514→    } finally {
   515→      setSaving(false);
   516→    }
   517→  }
   518→
   519→  if (!isNew && isLoading) {
   520→    return (
   521→      <div className="space-y-4">
   522→        <Skeleton className="h-10 w-64" />
   523→        <Skeleton className="h-96 rounded-xl" />
   524→      </div>
   525→    );
   526→  }
   527→
   528→  return (
   529→    <div className="space-y-4">
   530→      <div className="flex flex-wrap items-center gap-2">
   531→        <Button variant="ghost" size="sm" onClick={onBack}>
   532→          <ArrowLeft className="mr-1 h-4 w-4" /> All blog posts
   533→        </Button>

   535→        <BlogStatusBadge status={form.status} />
   536→      </div>
   537→
   538→      <Tabs value={tab} onValueChange={setTab}>
   539→        <TabsList className="flex-wrap h-auto">
   540→          <TabsTrigger value="content">Content</TabsTrigger>
   541→          <TabsTrigger value="publish">Publish</TabsTrigger>
   542→          <TabsTrigger value="seo">SEO</TabsTrigger>
   543→          <TabsTrigger value="relations">Relations & FAQs</TabsTrigger>
   544→        </TabsList>
   545→
   546→        {/* -------- Content -------- */}

   553→            <Textarea rows={2} value={form.excerpt} onChange={(e) => set("excerpt", e.target.value)} />

   556→            <RichTextEditor value={form.content} onChange={(v) => set("content", v)} rows={16} variant="full" internalLinks={internalLinks} />

   559→            <MediaPickField
   560→              value={form.coverImage}

   562→              onPick={(pick) => {

   564→                set("coverAlt", pick.alt);
   565→              }}
   566→              onClear={() => {
   567→                set("coverImage", "");
   568→                set("coverAlt", "");
   569→              }}
   570→            />
   571→          </Field>
   572→          <Field label="Cover alt text">
   573→            <Input value={form.coverAlt} onChange={(e) => set("coverAlt", e.target.value)} />
   574→          </Field>
   575→          <div className="grid gap-4 md:grid-cols-2">
   576→            <Field label="Author">
   577→              <Input value={form.author} onChange={(e) => set("author", e.target.value)} />
   578→            </Field>
   579→            <Field label="Reading time" hint="Auto-calculated from the content.">
   580→              <div className="flex items-center gap-2">
   581→                <Input type="number" value={form.readingTime} onChange={(e) => set("readingTime", Number(e.target.value) || 0)} className="w-24" />
   582→                <span className="text-sm text-muted-foreground">min</span>
   583→                <Button type="button" variant="outline" size="sm" onClick={estimateReadingTime}>
   584→                  ✨ Auto-calculate
   585→                </Button>

   587→            </Field>
   588→          </div>
   589→          <div className="grid gap-4 md:grid-cols-2">
   590→            <Field label="Tags" hint="Comma separated.">
   591→              <Input value={form.tags} onChange={(e) => set("tags", e.target.value)} placeholder="resin care, gifting ideas" />
   592→            </Field>
   593→            <Field label="Blog category">
   594→              <Select value={form.blogCategoryId} onValueChange={(v) => set("blogCategoryId", v === "__none" ? "" : v)}>
   595→                <SelectTrigger>
   596→                  <SelectValue placeholder="— none —" />
   597→                </SelectTrigger>
   598→                <SelectContent>
   599→                  <SelectItem value="__none">— none —</SelectItem>
   600→                  {(blogCategories ?? []).map((c) => (
   601→                    <SelectItem key={c.id} value={c.slug}>
   602→                      {c.name}
   603→                    </SelectItem>
   604→                  ))}
   605→                </SelectContent>
   606→              </Select>
   607→            </Field>
   608→          </div>
   609→          <div className="flex items-center gap-2">

   612→          </div>
   613→        </TabsContent>
   614→
   615→        {/* -------- Publish -------- */}
   616→        <TabsContent value="publish" className="mt-4 space-y-4">

   621→                  value={form.status}
   622→                  onValueChange={(v) => {

   626→                      set("publishAt", toDatetimeLocal(d.toISOString()));

   630→                  <SelectTrigger className="w-56">
   631→                    <SelectValue />

   633→                  <SelectContent>

   645→              {post?.publishedAt ? (

   650→              <p className="text-xs text-muted-foreground">

   657→
   658→        {/* -------- SEO -------- */}
   659→        <TabsContent value="seo" className="mt-4 space-y-4">
   660→          <div className="flex items-center justify-between gap-2">

   662→            <Button type="button" variant="outline" size="sm" onClick={generateSeoDefaults}>
   663→              ✨ Generate defaults
   664→            </Button>
   665→          </div>
   666→          <GooglePreview
   667→            title={form.seoTitle || form.title || "Blog post title"}
   668→            url={`artisticbykhushi.com/blog/${form.slug || "your-slug"}`}
   669→            description={form.metaDescription || form.excerpt}
   670→          />
   671→          <Field label="SEO title" counter={<CharCount text={form.seoTitle} target={60} />}>
   672→            <Input value={form.seoTitle} onChange={(e) => set("seoTitle", e.target.value)} />
   673→          </Field>
   674→          <Field label="Meta description" counter={<CharCount text={form.metaDescription} target={160} />}>
   675→            <Textarea rows={2} value={form.metaDescription} onChange={(e) => set("metaDescription", e.target.value)} />

   677→          <div className="grid gap-4 md:grid-cols-2">
   678→            <Field label="Focus keyword">
   679→              <Input value={form.focusKeyword} onChange={(e) => set("focusKeyword", e.target.value)} />
   680→            </Field>
   681→            <Field label="Secondary keywords" hint="Comma separated">
   682→              <Input value={form.secondaryKeywords} onChange={(e) => set("secondaryKeywords", e.target.value)} />
   683→            </Field>
   684→            <Field label="OG title">
   685→              <Input value={form.ogTitle} onChange={(e) => set("ogTitle", e.target.value)} />
   686→            </Field>

   688→              <Input value={form.canonicalUrl} onChange={(e) => set("canonicalUrl", e.target.value)} placeholder="https://…" />
   689→            </Field>
   690→          </div>
   691→          <Field label="OG description">
   692→            <Textarea rows={2} value={form.ogDescription} onChange={(e) => set("ogDescription", e.target.value)} />
   693→          </Field>
   694→        </TabsContent>

   696→        {/* -------- Relations -------- */}

   698→          <Field label="Related products" hint="Shown under this post as “You may also like”.">

   702→              </div>
   703→              <div className="max-h-64 overflow-y-auto custom-scroll p-2 space-y-1">

   714→                            checked

   716→                              : form.relatedProductSlugs.filter((s) => s !== p.slug),
   717→                          )
   718→                        }
   719→                      />
   720→                      <ImageThumb src={p.featuredImageUrl} alt="" size={24} />
   721→                      <span className="truncate">{p.name}</span>
   722→                    </label>
   723→                  ))
   724→                )}
   725→              </div>
   726→            </div>
   727→          </Field>
   728→
   729→          <Field label="Post FAQs" hint="FAQ block shown on this blog post page.">
   730→            <div className="space-y-2">
   731→              {form.faqs.length === 0 ? <p className="text-sm text-muted-foreground">No FAQs yet.</p> : null}
   732→              {form.faqs.map((faq, idx) => (
   733→                <div key={idx} className="rounded-lg border p-3 space-y-2">
   734→                  <div className="flex items-start gap-2">
   735→                    <Field label={`Question ${idx + 1}`} className="flex-1">

   737→                    </Field>
   738→                    <Button type="button" variant="ghost" size="icon" className="mt-6 text-destructive" onClick={() => set("faqs", form.faqs.filter((_, i) => i !== idx))} aria-label="Remove FAQ">
   739→                      <Trash2 className="h-4 w-4" />
   740→                    </Button>
   741→                  </div>
   742→                  <Field label="Answer">
   743→                    <Textarea rows={2} value={faq.answer} onChange={(e) => set("faqs", form.faqs.map((f, i) => (i === idx ? { ...f, answer: e.target.value } : f)))} />
   744→                  </Field>
   745→                </div>
   746→              ))}
   747→              <Button type="button" variant="outline" onClick={() => set("faqs", [...form.faqs, { question: "", answer: "" }])}>
   748→                <Plus className="mr-1 h-4 w-4" /> Add FAQ
   749→              </Button>
   750→              {blogFaqs.length > 0 ? (
   751→                <p className="text-xs text-muted-foreground">{blogFaqs.length} FAQ{blogFaqs.length === 1 ? "" : "s"} currently live on this post.</p>
   752→              ) : null}
   753→            </div>
   754→          </Field>
   755→        </TabsContent>
   756→      </Tabs>
   757→
   758→      {/* sticky save bar */}
   759→      <div className="sticky bottom-16 z-20 mt-6 rounded-xl border bg-card/95 p-3 shadow-lg backdrop-blur md:bottom-4">
   760→        <div className="flex items-center justify-between gap-2">
   761→          <p className="text-xs text-muted-foreground">{isNew ? "New post — not saved yet" : `Editing “${form.title}”`}</p>
   762→          <div className="flex gap-2">
   763→            <Button variant="outline" onClick={onBack} disabled={saving}>
   764→              Cancel
   765→            </Button>
   766→            <Button onClick={save} disabled={saving}>
   767→              {saving ? <Spinner className="mr-1" /> : <Save className="mr-1 h-4 w-4" />}
   768→              {saving ? "Saving…" : "Save post"}
   769→            </Button>
   770→          </div>
   771→        </div>
   772→      </div>
   773→    </div>
   774→  );
   775→}

   777→function safeParseSlugs(raw: string): string[] {
   778→  try {
   779→    const arr = JSON.parse(raw || "[]");
   780→    return Array.isArray(arr) ? arr.map(String) : [];
   781→  } catch {
   782→    return [];
   783→  }
