   150→    api
   151→      .put(`/api/admin/products/${product.id}`, { [field]: nextVal })

   153→        toast({
   154→          title: field === "published" ? (nextVal ? "Product published" : "Moved to drafts") : nextVal ? "Marked as featured" : "Removed from featured",

   157→        void qc.invalidateQueries({ queryKey: ["admin", "dashboard"] });

   246→          icon={<Package className="h-8 w-8 text-muted-foreground/50" />}

   250→            <Button className="mt-2" onClick={() => { setEditingId(null); setView("form"); }}>

   283→                    </TableCell>
   284→                    <TableCell>
   285→                      <Badge variant="secondary" className="max-w-36 truncate">{p.categoryName}</Badge>

   370→          <PaginationBar page={data?.page ?? 1} totalPages={data?.totalPages ?? 1} total={data?.total ?? 0} onPage={setPage} />

   379→        confirmLabel="Delete product"
   380→        onConfirm={confirmDelete}
   381→        pending={deleting}
   382→      />

   391→function ProductForm({ id, onBack }: { id: string | null; onBack: () => void }) {
   392→  const [currentId, setCurrentId] = useState<string | null>(id);
   393→  const isNew = !currentId;
   394→  const { data: product, isLoading } = useAdminProduct(currentId);
   395→  const { data: categories } = useAdminCategories();
   396→  const [form, setForm] = useState<ProductFormState>(EMPTY_FORM);
   397→  const [initialFaqIds, setInitialFaqIds] = useState<string[]>([]);
   398→  const [submitted, setSubmitted] = useState(false);
   399→  const [saving, setSaving] = useState(false);
   400→  const [tab, setTab] = useState("basics");
   401→  const [pickerOpen, setPickerOpen] = useState(false);
   402→  const [relatedQ, setRelatedQ] = useState("");
   403→  const initRef = useRef<string | null>(null);
   404→  const { toast } = useToast();
   405→  const qc = useQueryClient();
   406→
   407→  // FAQ list for this product (to diff on save)
   408→  const { data: productFaqs } = useAdminFaqs("");
   409→  const productFaqList = useMemo(
   410→    () => (productFaqs ?? []).filter((f) => f.entityType === "PRODUCT" && f.entityId === currentId),
   411→    [productFaqs, currentId],
   412→  );
   413→
   414→  // load existing product into form once
   415→  useEffect(() => {
   416→    if (!product || initRef.current === product.id) return;
   417→    initRef.current = product.id;
   418→    const faqs = (productFaqs ?? [])
   419→      .filter((f) => f.entityType === "PRODUCT" && f.entityId === product.id)
   420→      .map((f) => ({ id: f.id, question: f.question, answer: f.answer }));
   421→    setInitialFaqIds((productFaqs ?? []).filter((f) => f.entityType === "PRODUCT" && f.entityId === product.id).map((f) => f.id));
   422→    setForm({
   423→      name: product.name,
   424→      slug: product.slug,
   425→      sku: product.sku,
   426→      categoryId: product.categoryId,
   427→      subcategory: product.subcategory,
   428→      shortDescription: product.shortDescription,
   429→      longDescription: product.longDescription,
   430→      highlights: product.highlights,
   431→      customizationOptions: product.customizationOptions,
   432→      tags: product.tags,
   433→      images: product.gallery.map((g) => ({ ...g })),
   434→      size: product.size,
   435→      material: product.material,
   436→      colour: product.colour,
   437→      occasion: product.occasion,
   438→      careInstructions: product.careInstructions,
   439→      seoTitle: product.seoTitle,
   440→      metaDescription: product.metaDescription,
   441→      focusKeyword: product.focusKeyword,
   442→      secondaryKeywords: product.secondaryKeywords,
   443→      ogTitle: product.ogTitle,
   444→      ogDescription: product.ogDescription,
   445→      canonicalUrl: product.canonicalUrl,
   446→      faqs,
   447→      published: product.published,
   448→      featured: product.featured,
   449→      displayOrder: product.displayOrder,
   450→      relatedProductIds: safeParseIds(product.relatedProductIds),
   451→    });
   452→  }, [product, productFaqs]);
   453→
   454→  function set<K extends keyof ProductFormState>(key: K, value: ProductFormState[K]) {
   455→    setForm((f) => ({ ...f, [key]: value }));
   456→  }
   457→
   458→  // all products for related picker
   459→  const { data: allProducts } = useAdminProducts({ page: 1, pageSize: 200, q: "", category: "", status: "all" });
   460→  const relatedOptions = useMemo(() => {
   461→    const list = (allProducts?.items ?? []).filter((p) => p.id !== currentId);
   462→    const needle = relatedQ.trim().toLowerCase();
   463→    return needle ? list.filter((p) => p.name.toLowerCase().includes(needle) || p.sku.toLowerCase().includes(needle)) : list;
   464→  }, [allProducts, currentId, relatedQ]);
   465→
   466→  const nameError = submitted && !form.name.trim() ? "Product name is required." : undefined;
   467→  const categoryError = submitted && !form.categoryId ? "Please choose a category." : undefined;
   468→
   469→  // ---------- images ----------
   470→  function addImage(url: string, alt: string) {
   471→    setForm((f) => ({
   472→      ...f,
   473→      images: [
   474→        ...f.images,
   475→        { id: undefined, url, alt, caption: "", isFeatured: f.images.length === 0, displayOrder: f.images.length },
   476→      ],
   477→    }));
   478→  }
   479→
   480→  function updateImage(idx: number, patch: Partial<ImageRow>) {
   481→    setForm((f) => ({ ...f, images: f.images.map((img, i) => (i === idx ? { ...img, ...patch } : img)) }));
   482→  }
   483→
   484→  function removeImage(idx: number) {
   485→    setForm((f) => {
   486→      const images = f.images.filter((_, i) => i !== idx);
   487→      if (images.length > 0 && !images.some((i) => i.isFeatured)) images[0].isFeatured = true;
   488→      return { ...f, images };
   489→    });
   490→  }
   491→
   492→  function moveImage(idx: number, dir: -1 | 1) {
   493→    setForm((f) => {
   494→      const images = [...f.images];
   495→      const target = idx + dir;
   496→      if (target < 0 || target >= images.length) return f;
   497→      [images[idx], images[target]] = [images[target], images[idx]];
   498→      images.forEach((img, i) => (img.displayOrder = i));
   499→      return { ...f, images };
   500→    });
   501→  }
   502→
   503→  function setFeaturedImage(idx: number) {
   504→    setForm((f) => ({ ...f, images: f.images.map((img, i) => ({ ...img, isFeatured: i === idx })) }));
