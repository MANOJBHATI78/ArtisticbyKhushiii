    43→/** Editable {label, href} list stored as JSON in a settings key ("" = defaults). */
    44→function LinkListEditor({
    45→  value,
    46→  onChange,
    47→  placeholderHref,
    48→  defaults,
    49→}: {
    50→  value: string;
    51→  onChange: (json: string) => void;
    52→  placeholderHref: string;
    53→  defaults: NavLink[];
    54→}) {
    55→  const links: NavLink[] = useMemo(() => {
    56→    if (!value.trim()) return [];
    57→    try {
    58→      const parsed = JSON.parse(value);
    59→      return Array.isArray(parsed) ? parsed.filter((x) => x && typeof x.label === "string" && typeof x.href === "string") : [];
    60→    } catch {
    61→      return [];
    62→    }
    63→  }, [value]);
    64→
    65→  function commit(next: NavLink[]) {
    66→    onChange(next.length === 0 ? "" : JSON.stringify(next));
    67→  }
    68→  function update(i: number, patch: Partial<NavLink>) {
    69→    commit(links.map((l, idx) => (idx === i ? { ...l, ...patch } : l)));
    70→  }
    71→  function move(i: number, dir: -1 | 1) {
    72→    const j = i + dir;
    73→    if (j < 0 || j >= links.length) return;
    74→    const next = [...links];
    75→    [next[i], next[j]] = [next[j], next[i]];
    76→    commit(next);
    77→  }
    78→
    79→  return (
    80→    <div className="space-y-2">
    81→      {links.map((l, i) => (
    82→        <div key={i} className="flex items-center gap-1.5">
    83→          <Input
    84→            value={l.label}
    85→            onChange={(e) => update(i, { label: e.target.value })}
    86→            placeholder="Label"
    87→            className="h-9 w-32 shrink-0"
    88→            aria-label={`Link ${i + 1} label`}
    89→          />
    90→          <Input
    91→            value={l.href}
    92→            onChange={(e) => update(i, { href: e.target.value })}
    93→            placeholder={placeholderHref}
    94→            className="h-9"
    95→            aria-label={`Link ${i + 1} URL`}
    96→          />
    97→          <div className="flex shrink-0 items-center">
    98→            <Button type="button" variant="ghost" size="icon" className="size-8" onClick={() => move(i, -1)} disabled={i === 0} aria-label="Move up">
    99→              <ArrowUp className="size-3.5" />
   100→            </Button>
   101→            <Button type="button" variant="ghost" size="icon" className="size-8" onClick={() => move(i, 1)} disabled={i === links.length - 1} aria-label="Move down">
   102→              <ArrowDown className="size-3.5" />
   103→            </Button>
   104→            <Button type="button" variant="ghost" size="icon" className="size-8 text-destructive hover:text-destructive" onClick={() => commit(links.filter((_, idx) => idx !== i))} aria-label="Delete link">
   105→              <Trash2 className="size-3.5" />
   106→            </Button>
   107→          </div>
   108→        </div>
   109→      ))}
   110→      <Button
   111→        type="button"
   112→        variant="outline"
   113→        size="sm"
   114→        className="mt-1"
   115→        onClick={() => commit([...links, { label: "New link", href: "/" }])}
   116→      >
   117→        + Add link
   118→      </Button>
   119→      {links.length === 0 ? (
   120→        <div className="space-y-1.5">
   121→          <p className="text-xs text-muted-foreground">Empty = the built-in default links are shown on the site.</p>
   122→          <Button type="button" variant="ghost" size="sm" className="text-primary" onClick={() => commit(defaults)}>
   123→            Start from the current links
   124→          </Button>
   125→        </div>
   126→      ) : (
   127→        <Button type="button" variant="ghost" size="sm" className="text-destructive hover:text-destructive" onClick={() => commit([])}>
   128→          Reset to defaults
   129→        </Button>
   130→      )}
   131→    </div>
   132→  );
   133→}

   143→/** ISO string → value for <input type="datetime-local"> (local time). */
   144→function isoToLocalInput(iso: string): string {
   145→  if (!iso) return "";
   146→  const d = new Date(iso);
   147→  if (Number.isNaN(d.getTime())) return "";
   148→  const pad = (n: number) => String(n).padStart(2, "0");
   149→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
   150→}
   151→
   152→/** datetime-local input value → ISO string for storage. */
   153→function localInputToIso(local: string): string {
   154→  if (!local) return "";
   155→  const d = new Date(local);
   156→  return Number.isNaN(d.getTime()) ? "" : d.toISOString();
   157→}
