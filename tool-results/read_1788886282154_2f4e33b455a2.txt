   391→   391→function ProductForm({ id, onBack }: { id: string | null; onBack: () => void }) {
   392→   392→  const [currentId, setCurrentId] = useState<string | null>(id);
   393→   393→  const isNew = !currentId;
   394→   394→  const { data: product, isLoading } = useAdminProduct(currentId);
   395→   395→  const { data: categories } = useAdminCategories();
   396→   396→  const [form, setForm] = useState<ProductFormState>(EMPTY_FORM);
   397→   397→  const [initialFaqIds, setInitialFaqIds] = useState<string[]>([]);
   398→   398→  const [submitted, setSubmitted] = useState(false);
   399→   399→  const [saving, setSaving] = useState(false);
   400→   400→  const [tab, setTab] = useState("basics");
   401→   401→  const [pickerOpen, setPickerOpen] = useState(false);
   402→   402→  const [relatedQ, setRelatedQ] = useState("");
   403→   403→  const initRef = useRef<string | null>(null);
   404→   404→  const { toast } = useToast();
   405→   405→  const qc = useQueryClient();
   406→   406→
   407→   407→  // FAQ list for this product (to diff on save)
   408→   408→  const { data: productFaqs } = useAdminFaqs("");
   409→   409→  const productFaqList = useMemo(
   410→   410→    () => (productFaqs ?? []).filter((f) => f.entityType === "PRODUCT" && f.entityId === currentId),
   411→   411→    [productFaqs, currentId],
   412→   412→  );
   413→   413→
   414→   414→  // load existing product into form once
   415→   415→  useEffect(() => {
   416→   416→    if (!product || initRef.current === product.id) return;
   417→   417→    initRef.current = product.id;
   418→   418→    const faqs = (productFaqs ?? [])
   419→   419→      .filter((f) => f.entityType === "PRODUCT" && f.entityId === product.id)
   420→   420→      .map((f) => ({ id: f.id, question: f.question, answer: f.answer }));
   421→   421→    setInitialFaqIds((productFaqs ?? []).filter((f) => f.entityType === "PRODUCT" && f.entityId === product.id).map((f) => f.id));
   422→   422→    setForm({
   423→   423→      name: product.name,
