   412→function ProductForm({ id, onBack }: { id: string | null; onBack: () => void }) {
   413→  const [currentId, setCurrentId] = useState<string | null>(id);
   414→  const isNew = !currentId;
   415→  const { data: product, isLoading } = useAdminProduct(currentId);
   416→  const { data: categories } = useAdminCategories();
   417→  const [form, setForm] = useState<ProductFormState>(EMPTY_FORM);
   418→  const [initialFaqIds, setInitialFaqIds] = useState<string[]>([]);
   419→  const [submitted, setSubmitted] = useState(false);
   420→  const [saving, setSaving] = useState(false);
   421→  const [tab, setTab] = useState("basics");
   422→  const [pickerOpen, setPickerOpen] = useState(false);
   423→  const [relatedQ, setRelatedQ] = useState("");
   424→  const initRef = useRef<string | null>(null);
   425→  const { toast } = useToast();
   426→  const qc = useQueryClient();
   427→
   428→  // FAQ list for this product (to diff on save)
   429→  const { data: productFaqs } = useAdminFaqs("");
   430→  const productFaqList = useMemo(
   431→    () => (productFaqs ?? []).filter((f) => f.entityType === "PRODUCT" && f.entityId === currentId),
   432→    [productFaqs, currentId],
   433→  );
   434→
   435→  // load existing product into form once (waits for both product and the
   436→  // FAQ list so the FAQs tab is pre-filled without a race)
   437→  useEffect(() => {
   438→    if (!product || !productFaqs || initRef.current === product.id) return;
   439→    initRef.current = product.id;
   440→    const faqs = productFaqs
   441→      .filter((f) => f.entityType === "PRODUCT" && f.entityId === product.id)
   442→      .map((f) => ({ id: f.id, question: f.question, answer: f.answer }));
   443→    setInitialFaqIds(faqs.map((f) => f.id as string));
   444→    setForm({
   445→      name: product.name,
   446→      slug: product.slug,
   447→      sku: product.sku,
   448→      categoryId: product.categoryId,
   449→      subcategory: product.subcategory,
   450→      shortDescription: product.shortDescription,
   451→      longDescription: product.longDescription,
   452→      highlights: product.highlights,
   453→      customizationOptions: product.customizationOptions,
   454→      tags: product.tags,
   455→      images: product.gallery.map((g) => ({ ...g })),
   456→      size: product.size,
   457→      material: product.material,
   458→      colour: product.colour,
   459→      occasion: product.occasion,
   460→      careInstructions: product.careInstructions,
   461→      seoTitle: product.seoTitle,
   462→      metaDescription: product.metaDescription,
   463→      focusKeyword: product.focusKeyword,
   464→      secondaryKeywords: product.secondaryKeywords,
   465→      ogTitle: product.ogTitle,
   466→      ogDescription: product.ogDescription,
   467→      canonicalUrl: product.canonicalUrl,
   468→      faqs,
   469→      published: product.published,
   470→      featured: product.featured,
   471→      displayOrder: product.displayOrder,
   472→      relatedProductIds: safeParseIds(product.relatedProductIds),
   473→    });
   474→  }, [product, productFaqs]);
   475→
   476→  function set<K extends keyof ProductFormState>(key: K, value: ProductFormState[K]) {
   477→    setForm((f) => ({ ...f, [key]: value }));
   478→  }
   479→
   480→  // all products for related picker
   481→  const { data: allProducts } = useAdminProducts({ page: 1, pageSize: 200, q: "", category: "", status: "all" });
   482→  const relatedOptions = useMemo(() => {
   483→    const list = (allProducts?.items ?? []).filter((p) => p.id !== currentId);
   484→    const needle = relatedQ.trim().toLowerCase();
   485→    return needle ? list.filter((p) => p.name.toLowerCase().includes(needle) || p.sku.toLowerCase().includes(needle)) : list;
   486→  }, [allProducts, currentId, relatedQ]);
   487→
   488→  const nameError = submitted && !form.name.trim() ? "Product name is required." : undefined;
   489→  const categoryError = submitted && !form.categoryId ? "Please choose a category." : undefined;
   490→
   491→  // ---------- images ----------
   492→  function addImage(url: string, alt: string) {
   493→    setForm((f) => ({
   494→      ...f,
   495→      images: [
   496→        ...f.images,
   497→        { id: undefined, url, alt, caption: "", isFeatured: f.images.length === 0, displayOrder: f.images.length },
   498→      ],
   499→    }));

   947→                          )
   948→                        }
   949→                      />
   950→                      <ImageThumb src={p.featuredImageUrl} alt="" size={24} />

   954→                )}
   955→              </div>

   960→
   961→      {/* sticky save bar */}
   962→      <div className="sticky bottom-16 z-20 mt-6 rounded-xl border bg-card/95 p-3 shadow-lg backdrop-blur md:bottom-4">
   963→        <div className="flex items-center justify-between gap-2">
   964→          <p className="text-xs text-muted-foreground">
   965→            {isNew ? "New product — not saved yet" : `Editing “${form.name}”`}
   966→          </p>
   967→          <div className="flex gap-2">
   968→            <Button variant="outline" onClick={onBack} disabled={saving}>
   969→              Cancel
   970→            </Button>
   971→            <Button onClick={save} disabled={saving}>
   972→              {saving ? <Spinner className="mr-1" /> : <Save className="mr-1 h-4 w-4" />}

   974→            </Button>
